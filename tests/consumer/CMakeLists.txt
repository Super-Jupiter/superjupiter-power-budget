cmake_minimum_required(VERSION 3.16)
project(SuperJupiterPowerBudgetConsumer LANGUAGES CXX)
if(DEFINED SUPERJUPITER_POWER_BUDGET_SOURCE_DIR)
  add_subdirectory("${SUPERJUPITER_POWER_BUDGET_SOURCE_DIR}" power-budget)
  if(TARGET superjupiter-power-budget-tests OR TARGET superjupiter-power-budget-replay)
    message(FATAL_ERROR "A consumer unexpectedly builds library developer tools")
  endif()
else()
  find_package(SuperJupiterPowerBudget 0.1 CONFIG REQUIRED)
endif()
add_executable(superjupiter-power-budget-consumer main.cpp)
target_link_libraries(superjupiter-power-budget-consumer PRIVATE SuperJupiter::PowerBudget)
